6. COUNT MILESTONE COMPLETIONS
═══════════════════════════════════════════════════════════════════════════════

fetch('/api/companies')
.then(r => r.json())
.then(companies => {
  const milestones = {
    'Order Started': 0,
    'Agent Assigned': 0,
    'Address Issued': 0,
    'Application Applied': 0,
    'Formation Complete': 0,
    'EIN Application': 0,
    'EIN Obtained': 0
  }
  
  companies.forEach(c => {
    const m = c.milestones || {}
    if (m.orderSuccessfullyProcessed) milestones['Order Started']++
    if (m.registeredAgentAssigned) milestones['Agent Assigned']++
    if (m.businessMailingAddressIssued) milestones['Address Issued']++
    if (m.companyApplicationApplied) milestones['Application Applied']++
    if (m.companyFormationCompleted) milestones['Formation Complete']++
    if (m.einApplicationSubmitted) milestones['EIN Application']++
    if (m.einObtained) milestones['EIN Obtained']++
  })
  
  console.table(milestones)
})

7. EXPORT MILESTONES TO CSV
═══════════════════════════════════════════════════════════════════════════════

fetch('/api/companies')
.then(r => r.json())
.then(companies => {
  let csv = 'Company ID,Name,Order Started,Agent Assigned,Address,Application Applied,Formation,EIN App,EIN\n'
  
  companies.forEach(c => {
    const m = c.milestones || {}
    csv += `"${c.id}","${c.name}",${m.orderSuccessfullyProcessed ? '✓' : ''},${m.registeredAgentAssigned ? '✓' : ''},${m.businessMailingAddressIssued ? '✓' : ''},${m.companyApplicationApplied ? '✓' : ''},${m.companyFormationCompleted ? '✓' : ''},${m.einApplicationSubmitted ? '✓' : ''},${m.einObtained ? '✓' : ''}\n`
  })
  
  const blob = new Blob([csv], { type: 'text/csv' })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `milestones-${new Date().toISOString().split('T')[0]}.csv`
  a.click()
})

8. MONITOR MIGRATION PROGRESS (Run this while migration is running)
═══════════════════════════════════════════════════════════════════════════════

let previousCount = 0
const monitor = setInterval(async () => {
  const response = await fetch('/api/companies')
  const companies = await response.json()
  const count = companies.filter(c => c.milestones?.companyApplicationApplied).length
  
  if (count !== previousCount) {
    console.log(`[${new Date().toLocaleTimeString()}] Progress: ${count} companies updated`)
    previousCount = count
  }
  
  if (count === companies.length) {
    console.log('✓ MIGRATION COMPLETE!')
    clearInterval(monitor)
  }
}, 1000)
